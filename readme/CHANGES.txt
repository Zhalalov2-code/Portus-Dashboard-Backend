Field: service
  Type: TINYINT(1) — returned as int (0 or 1)
  Default: 0
  Accepted in: POST /lkw, PUT /lkw/{id}, POST /chassi, PUT /chassi/{id}

  { "id_lkw": 12, "service": 1 }
